# The names and numbers that the agent and a phone app must both have, and have the same (see the plan document,
# "Phone and agent protocol" and "Bluetooth serial on the car"). Each side holds them in its own code and has a
# test that reads this file and fails when the two differ: the agent's is tests/agent/test_constants.cpp, against
# agent/protocol_constants.hpp.
#
# One to a line: a name, a space, the value. A line that starts with # is a comment.

# The one version of the protocol there is: what `hello` offers and answers.
protocol.version 1

# A frame's payload in bytes: the most there is, and the least `hello` may ask for.
frame.maxPayload 65536
frame.minPayload 1024

# Bluetooth serial. The service the agent registers with the car's Bluetooth stack: the name its registration is
# filed under on the car, and the UUID a phone app connects to.
serial.serviceName NBTApp
serial.serviceUuid 58876F35-7F3F-4592-B65D-9912589A4F51

# On a serial link each side gives this, at most, as its maxFrame in `hello`.
serial.maxFrame 4096

# On a serial link, each side: when it has heard nothing for the first of these, in milliseconds, it sends `ping`,
# and again each time as long has passed with nothing heard; when it has heard nothing for the second, and has
# sent those pings, it ends the link. Anything that arrives counts as heard. The agent also ends a serial link on
# which no `hello` has been done when the second has passed since the link opened.
serial.pingAfterMs 10000
serial.silenceMs 30000

# The car gives every link a phone opens to whichever program reads it first, so what the phone says first decides
# whose link it is. A phone app sends the magic bytes as soon as its link is open. The agent waits this long for
# them, in milliseconds from when it opened the link, after any whole frames that an earlier link left unread, and
# lets a link go on which they have not come by then, or on which anything else came first: it writes nothing into
# such a link.
serial.decideMs 500

# The phone made the link, and the phone ends it. When the agent ends a link on purpose it sends the event
# `link.closing` first, and a phone app closes its side of the link as soon as that event arrives, sending nothing
# more. The agent keeps the link's node open and reads it, dropping what comes, until the phone has done so: closing
# a node does not end its link, and what a phone sends into a link whose node was closed comes ahead of the bytes of
# every later link. When the link is still up this long, in milliseconds from when the agent closed its own end,
# the agent's log says so.
serial.lingerMs 5000

# Pairing a phone (the plan document, "Enrolment on the car's screen"). The car's screen shows a code of this many
# digits, which the user types into the phone app; `pairing.issue` carries it.
pairing.codeDigits 6

# The code is good for the first of these, in seconds, from when the screen asked for it; that is the `expiresIn`
# of `pairing.open`. The agent takes this many wrong codes, the last of which closes the window, and for the last
# of the three values, in seconds, it then opens no new one.
pairing.windowSeconds 120
pairing.tries 3
pairing.pauseSeconds 30

# `auth`: each side's nonce, in bytes. It travels as twice as many hex characters.
auth.nonceBytes 16

# Capabilities: what a side has beyond the protocol's first version, by the names it lists in `hello`. An agent
# that lists this one has the methods autostart.get and autostart.set, which say and set whether the agent is
# started with the head unit; one that does not list it answers them with `unsupported`.
capability.autostart autostart
# An agent that lists this one installs an offer with the signature the offer gives (job.plan's `signature`), checked
# over the manifest, and does not read the archive's manifest.sig. That signature may be a catalogue's detached one,
# for a package whose author signed it with a key of their own. An agent that does not list it checks manifest.sig
# when it installs, so such a package is refused there with `bad-signature`.
capability.detachedSignature detached-signature
# An agent that lists this one has files.list, files.fetch and files.cancel: it sends a controller the files of an
# installed package, as yet only the logs its manifest names in `logs` and the agent's own (the Message reference,
# "Files"). One that does not list it answers them with `unsupported`.
capability.files files
# An agent that lists this one has checks.run and checks.fix: it looks for what was put onto the head unit by hand,
# outside this system, and stands in the way of the head unit (the system partition left writable, an app that was
# installed by hand), and puts it right when a controller asks, with the token of what the user was shown. One that
# does not list it answers them with `unsupported`.
capability.checks checks
# An agent that lists this one has serial.reset: it asks the car's Bluetooth service to end the phone's Bluetooth
# serial link, which closing the link's node does not do, at once if a serial link's node is open and otherwise at
# the open of the next one. Only an agent that takes Bluetooth serial links lists it; one that does not list it
# answers serial.reset with `unsupported`.
capability.serialReset serial-reset
# An agent that lists this one has the car dial a phone of its own: when the phone is connected to the car and
# paired with it, the car opens a serial link to the agent's service UUID on the phone, which the phone's app takes
# if it listens there. The link is the agent's as one the phone opened is: it opens with the phone's magic bytes,
# and `hello` follows. Only an agent whose Bluetooth serial listener dials lists it.
capability.serialDial serial-dial
# An agent that lists this one has settings.get and settings.set, which give and change the settings an installed
# package's manifest names in `settings` (spec/manifest.schema.json). One that does not list it answers them with
# `unsupported`.
capability.settings settings

# Package settings: the most bytes one setting's value has, as it stands in the package's settings file. A `text`
# option's maxLength is at most this.
settings.maxValueBytes 255

# An agent that lists this one has channels: a controller opens a named channel of an installed package with
# channel.open, and messages pass both ways as DATA frames on the channel's stream number, with credits
# (channel.credit), until either side closes it (channel.close); channel.get and channel.set say and set whether a
# switched channel is on. The agent connects to the package's program for each channel, and may start it
# (agent/agent.hpp has the messages). One that does not list it answers those methods with `unsupported`.
capability.channels channels

# Stream numbers. A files stream (files.fetch) and the stream of a blob.fetch take the numbers from 1 to the first of
# these, and start again at 1 after it; a channel takes the numbers from the second up to 65,535, and starts again
# at the second.
stream.last 32767
channel.firstStream 32768

# A channel's messages: one message is one DATA frame, with the last flag set, of 1 to this many bytes, on every
# link. A link whose frame does not hold that many has no channels.
channel.maxMessage 4096

# How many messages each side may send on a channel before the other grants it more with channel.credit: what the
# agent may send when channel.open names no credit, and what the answer to channel.open gives the phone.
channel.credit 16

# The most channels open on one link at a time.
channel.perLink 8

# A channel whose program the agent starts on demand: the agent tries to connect to it every second of these, in
# milliseconds, for the first, and answers channel.open with `unavailable` when the program has not taken the
# connection by then.
channel.startWaitMs 5000
channel.startRetryMs 100
